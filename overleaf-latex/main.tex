\slidesection{Motivation --- The Challenge of Membrane Characterization}
\textbf{Header:} Motivation: Why Intrusive UQ for Membranes?
\begin{itemize}[leftmargin=*]
  \item \textbf{The Problem:} Membrane transport parameters (permeability $B$, reflection $\sigma$) cannot be measured directly; they must be inferred from noise-corrupted, dynamic data.
  \item \textbf{The Gap:} Traditional steady-state experiments are slow and often fail to explore the concentration-dependent regions where parameters become identifiable.
  \item \textbf{The Opportunity:} By using Model-Based Design of Experiments (MBDoE), we can treat the experiment as a controllable dynamic system, choosing operating conditions that force the system into informative states.
\end{itemize}
\textbf{Key Takeaway:} Efficient characterization requires moving from trial-and-error testing to optimal information gathering.

\slidesection{General Forward Model (ODE-Constrained)}
\textbf{Header:} Mechanistic Foundation \& Notation
\begin{itemize}[leftmargin=*]
  \item System dynamics governed by nonlinear ODEs with parameters $\theta$.
  \item Experimental design $\phi$ parameterizes the time-varying inputs $u(t)$.
\end{itemize}
\[
\dot{x}(t) = f(x(t), u(t; \phi), \theta), \quad x(0) = x_0
\]
\[
\hat{y}^r(t) = g^r(x(t), u(t; \phi), \theta)
\]
\textbf{Key Takeaway:} The Forward Problem maps physical parameters and design choices to observable trajectories.

\slidesection{Measurement \& Noise Model}
\textbf{Header:} Statistical Characterization of Data
\begin{itemize}[leftmargin=*]
  \item Data = model truth + stochastic noise.
  \item Heterogeneous sensors (e.g., flux vs. concentration) require weighted objectives.
\end{itemize}
\[
y^r(t) = \hat{y}^r(t; \theta, \phi) + \epsilon^r(t), \quad \epsilon^r \sim \mathcal{N}(0, V_r)
\]
\textbf{Key Takeaway:} Explicitly modeling the noise covariance $V_r$ is essential for multi-modal data fusion.

\slidesection{Parameter Estimation (The Inverse Problem)}
\textbf{Header:} Weighted Least Squares (WLS) Formulation
\begin{itemize}[leftmargin=*]
  \item Estimate $\hat{\theta}$ by minimizing the Weighted Sum of Squared Errors (WSSE).
  \item This is the Maximum Likelihood Estimate (MLE) under the Gaussian assumption.
\end{itemize}
\[
\hat{\theta} = \arg\min_{\theta} \sum_r \left\lVert y^r - \hat{y}^r(\theta, \phi) \right\rVert_{V_r^{-1}}^2
\]
\textbf{Key Takeaway:} Parameter estimation is the inner loop of the UQ framework.

\slidesection{Intrusive Sensitivities (The Engine)}
\textbf{Header:} Exact Derivatives via Sensitivity ODEs
\begin{itemize}[leftmargin=*]
  \item Intrusive means differentiating the physics directly, not using finite differences.
  \item State sensitivity $S(t)$ propagates parameter influence through the dynamics.
\end{itemize}
\[
\dot{S}(t) = \frac{\partial f}{\partial x}S(t) + \frac{\partial f}{\partial \theta}, \quad S(0)=0
\]
\[
Q^r(t) = \frac{\partial g^r}{\partial x}S(t) + \frac{\partial g^r}{\partial \theta}
\]
\textbf{Key Takeaway:} Derivative-consistent sensitivities provide the high-fidelity gradients needed for the FIM.

\slidesection{Fisher Information \& Covariance}
\textbf{Header:} Mapping Sensitivity to Information Geometry
\begin{itemize}[leftmargin=*]
  \item FIM ($M$) summarizes the local information content of an experiment.
  \item Cram\'er-Rao Bound: $M^{-1}$ approximates the local parameter covariance.
\end{itemize}
\[
M(\hat{\theta},\phi) = \sum_r \int_0^T Q^r(t)^T V_r^{-1} Q^r(t)\,dt
\]
\[
\mathrm{Cov}(\hat{\theta}) \approx M(\hat{\theta},\phi)^{-1}
\]
\textbf{Key Takeaway:} The FIM quantifies how sensitive our measurements are to parameter changes.

\slidesection{Identifiability Diagnostics}
\textbf{Header:} Spectrum of the Information Matrix
\begin{itemize}[leftmargin=*]
  \item Condition number $\kappa(M)$ quantifies ill-conditioning and sloppy directions.
  \item Small eigenvalues $\lambda_{\min}$ indicate practically unidentifiable parameters.
\end{itemize}
\[
\kappa(M) = \frac{\lambda_{\max}(M)}{\lambda_{\min}(M)}
\]
\textbf{Key Takeaway:} Spectral analysis of $M$ shows which parameters the experiment failed to capture.

\slidesection{MBDoE Objective (D-Optimality)}
\textbf{Header:} Designing for Maximum Information
\begin{itemize}[leftmargin=*]
  \item Goal: Choose $\phi$ to minimize the volume of the uncertainty ellipsoid.
  \item D-optimality is invariant under linear reparameterization.
\end{itemize}
\[
\phi^* = \arg\max_{\phi} \log\det M(\hat{\theta}, \phi)
\]
\textbf{Key Takeaway:} Maximizing the determinant of the FIM is equivalent to maximizing Shannon information gain.

\slidesection{Sequential Design Workflow}
\textbf{Header:} Closing the Loop: Sequential MBDoE
\begin{itemize}[leftmargin=*]
  \item Update total information: $M_{\mathrm{total}} = M_{\mathrm{prior}} + M_{\mathrm{new}}$.
  \item Stop when the trace of covariance (A-optimality) meets tolerance $\tau$.
\end{itemize}
\[
\mathrm{tr}(M^{-1}) \leq \tau
\]
\textbf{Key Takeaway:} Sequential design reduces experimental effort by targeting only remaining uncertainty.
